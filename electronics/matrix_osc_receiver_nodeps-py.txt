#!/usr/bin/env python3
import socket
import struct
import sys
import threading
import time
import RPi.GPIO as GPIO

LISTEN_IP = "0.0.0.0"
LISTEN_PORT = 8000

COL_PINS = [12, 16, 18, 22, 32, 37]   # left -> right
ROW_PINS = [33, 31, 29, 15, 13, 11]   # top -> bottom

GPIO.setmode(GPIO.BOARD)
for p in COL_PINS + ROW_PINS:
    GPIO.setup(p, GPIO.OUT)
    GPIO.output(p, GPIO.LOW)

matrix_state = [[0] * 6 for _ in range(6)]
matrix_lock = threading.Lock()


def set_pattern_from_flat(flat_values):
    if len(flat_values) != 36:
        print(f"Expected 36 values, got {len(flat_values)} - ignoring frame", flush=True)
        return
    with matrix_lock:
        for r in range(6):
            for c in range(6):
                matrix_state[r][c] = 1 if flat_values[r * 6 + c] else 0


def _scan_loop():
    while True:
        with matrix_lock:
            snapshot = [row[:] for row in matrix_state]
        for r_idx, row_pin in enumerate(ROW_PINS):
            GPIO.output(row_pin, GPIO.HIGH)
            for c_idx, col_pin in enumerate(COL_PINS):
                GPIO.output(col_pin, GPIO.HIGH if snapshot[r_idx][c_idx] else GPIO.LOW)
            time.sleep(0.002)
            GPIO.output(row_pin, GPIO.LOW)
            for col_pin in COL_PINS:
                GPIO.output(col_pin, GPIO.LOW)


def start_matrix_scan():
    t = threading.Thread(target=_scan_loop, daemon=True)
    t.start()


def pad4(n):
    return (n + 3) & ~3


def read_osc_string(data, offset):
    end = data.index(b"\x00", offset)
    s = data[offset:end].decode("utf-8", errors="replace")
    new_offset = pad4(end + 1 - offset) + offset
    return s, new_offset


def parse_osc_message(data):
    offset = 0
    address, offset = read_osc_string(data, offset)

    if offset >= len(data) or data[offset:offset + 1] != b",":
        return address, []

    type_tags, offset = read_osc_string(data, offset)
    type_tags = type_tags[1:]

    args = []
    for tag in type_tags:
        if tag == "i":
            val = struct.unpack(">i", data[offset:offset + 4])[0]
            offset += 4
        elif tag == "f":
            val = struct.unpack(">f", data[offset:offset + 4])[0]
            offset += 4
        elif tag == "s":
            val, offset = read_osc_string(data, offset)
        elif tag == "b":
            blob_len = struct.unpack(">i", data[offset:offset + 4])[0]
            offset += 4
            val = data[offset:offset + blob_len]
            offset += pad4(blob_len)
        elif tag == "T":
            val = True
        elif tag == "F":
            val = False
        elif tag == "N":
            val = None
        else:
            val = None
        args.append(val)

    return address, args


def parse_packet(data, results):
    if data[:7] == b"#bundle":
        offset = 16
        while offset < len(data):
            elem_size = struct.unpack(">i", data[offset:offset + 4])[0]
            offset += 4
            elem_data = data[offset:offset + elem_size]
            offset += elem_size
            parse_packet(elem_data, results)
    else:
        address, args = parse_osc_message(data)
        results.append((address, args))


def handle_message(address, args):
    if address == "/pattern":
        set_pattern_from_flat(args)
        print(f"[OSC] pattern updated: {args}", flush=True)
    else:
        print(f"[OSC] address={address} args={args}", flush=True)


def main():
    start_matrix_scan()

    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    sock.bind((LISTEN_IP, LISTEN_PORT))
    print(f"OSC listener bound on UDP port {LISTEN_PORT}.", flush=True)

    try:
        while True:
            data, addr = sock.recvfrom(65535)
            try:
                results = []
                parse_packet(data, results)
                for address, args in results:
                    handle_message(address, args)
            except Exception as e:
                print(f"Failed to parse packet from {addr}: {e}", flush=True)
    except KeyboardInterrupt:
        print("\nShutting down.", flush=True)
    finally:
        GPIO.cleanup()
        sock.close()


if __name__ == "__main__":
    sys.exit(main())
